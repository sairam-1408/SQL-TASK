Write a stored procedure to update the salary of an employee using EmployeeID.
Create procedure ba(in emp_id int)
Begin 
Update employees set salary=salary+5000 where id=emp_id
End \\
Call ba(8) \\

Write a stored procedure to calculate the average salary of employees.
Create procedure ab2()
Begin
Select avg(salary)as average_salary from employees;
End \\
Call ab2() \\

Create a stored procedure that returns the total number of employees in a department.
Create procedure count(in dept varchar(50))
Begin
Select count(department) as total_number_of_employees from employees where department=dept;
End \\
Call count() \\


Write a stored procedure to find the maximum salary in the Employees table.
Create procedure max()
Begin
Select max(salary) as max_salary from employees;
End  \\
Call max() \\


Create a stored procedure that gives a bonus (10%) to employees whose salary is below 50,000.
Create procedure bonus()
Begin
Update employees set salary=salary+(salary*10/100 )where salary<50000;
End \\
Call bonus() \\
