INNER JOIN
1. Write a query to display employee name and department name.
Select e.name,d.department_name from employee  as e inner join departments as d on e.department_id =d.department_id;
2. Write a query to list employees working on projects.
Select e.name, p.project_name from employee as e inner join projects as p on e.project_id=p.project_id;
3. Write a query to find employees who are assigned to a department.
Select e.name,d.department_name from employee as e inner join department as d on e.department_id=d.department_id;
4. Display employee name and project name using INNER JOIN.
Select e.name,p.project_name from employee as e inner join project as p
On e.project_id=p.project_id;


LEFT JOIN 
5. Write a query to display all employees, including those without a department.
Select * from employees as e left join department as d on e.department_id=d.department_id;
6. Display all employees and their projects, including employees with no projects.
Select * from employees as e left join project as p on e.project_id=p.project_id;
7. Write a query to show employee name and department name, even if the department is NULL.
Select e.name,d.name from employee as e left join department as d on e.department_id =d.department_id;


RIGHT JOIN 
8. Write a query to display all departments, even if no employees work in them.
Select *from employees  as e right join department as d on e.department_id =d.department_id;
9. Display all projects, including those not assigned to any employee.
Select * from employee as e right join projects as p on e.project_id =p.project_id;
